<p>Theia exists for <a href="../../../Projects/Games/Sandtide/what-is-sandtide" class="internal alias" data-slug="Projects/Games/Sandtide/what-is-sandtide">Sandtide</a>. And while developing it, I’ve realized <strong>how wrong some stuff was…</strong> not performance-wise, but maintainability-wise. I understood Theia, but <span class="text-highlight">only because I created it</span>, and that’s a huge problem! A few times, with my colleague, I had to explain how to use it properly to avoid memory corruption errors and things alike, and honestly, <span class="text-highlight">that’s terrible</span>.</p>
<p>A framework <strong>should explicitly bring its philosophy along with it</strong>, and Theia did it, but partially. I had good ideas while developing it, but it was far from perfect; actually, <span class="text-highlight">the framework wasn’t self-aware of its own mistakes</span>. What I’m trying to say, is that the framework itself wasn’t enforcing anything, it led to a certain way of development, but it wasn’t explicitly acknowledging its use to the developer. <span class="text-highlight">It’s not only about having good documentation, but the system must react to wrong inputs</span>.</p>
<p>Performance is important, sure it is! But we need to address performance while being, simultaneously, explicit and constrained. And that’s what the new version of Theia is about… I’m still refining some edges before releasing it publicly, but I’ve come here to share the first benchmark results, comparing it to other awesome frameworks, all written in C#.</p>
<p><strong>Theia’s new version is fully implemented in managed C# (that means no unsafe code)</strong>, has a considerable amount of test coverage, and will be available with all the features from the previous version and more interesting ones!</p>
<p>I’ve decided to compile all the results and plot them on a graph, so it’s easier to visualize and digest them. I will release the result files along with Theia’s new version.</p>
<p>It’s safe to say that, by analyzing the data, <strong>Theia’s new version is not only safer, as I promised, but fast!</strong> In my opinion (try to ignore the fact that I’m the one who developed it), based on the data, Theia is a viable ECS framework. Due to its archetype’s underlying implementation, which aims for lazy initialization for chunks, at high entity counts, structural changes when adding or removing components are a little bit costly, and queries that involve very high entity counts can be impacted a little bit due to pointer dereferencing.</p>
<p>These weak spots are a tradeoff… these decisions allowed <strong>reduction of unused memory consumption, safer chunk access (especially for parallelization), unlimited component types, and other ins-and-outs.</strong> But no worries, Theia compensates its weaknesses a lot! As I like to point out, <span class="text-highlight">Theia is a fast-enough (and often, or sometimes, faster) ECS framework for your games!</span></p>

const FENNECS_IDX = 5;
let catIdx=0,tIdx=0,chart=null;

function buildChart(){
  const d=DATA[catIdx];
  const tv=d.t[tIdx];
  if(!tv)return;

  document.getElementById('sub').textContent=tv.sub;
  document.getElementById('ins').textContent=tv.ins;

  const isDark=matchMedia('(prefers-color-scheme: dark)').matches;
  const gc=isDark?'rgba(255,255,255,0.07)':'rgba(0,0,0,0.06)';
  const tc=isDark?'#999':'#666';

  const datasets=tv.rows.map((r,i)=>({
    label:FW[i].n,data:r,
    borderColor:FW[i].c,backgroundColor:FW[i].c+'22',
    borderWidth:i===0?2.5:1.5,
    pointRadius:i===0?4:2.5,
    hidden:!visible[i],
    spanGaps:true,tension:0.3,fill:false,
    borderDash:i===0?[]:[4,3],
  }));

  if(chart)chart.destroy();
  chart=new Chart(document.getElementById('ch'),{
    type:'line',data:{labels:L,datasets},
    options:{
      responsive:true,maintainAspectRatio:false,
      plugins:{legend:{display:false},tooltip:{callbacks:{label:c=>` ${c.dataset.label}: ${c.parsed.y!=null?c.parsed.y.toFixed(1)+' μs':'N/A'}`}}},
      scales:{
        x:{grid:{color:gc},ticks:{color:tc,font:{size:11}}},
        y:{grid:{color:gc},ticks:{color:tc,font:{size:11},callback:v=>v>=1000?(v/1000).toFixed(1)+'ms':v+'μs'}},
      }
    }
  });
}

function syncLegendUI(){
  const noFennecs = !!DATA[catIdx].noFennecs;
  document.querySelectorAll('.li').forEach((el,i)=>{
    const isNA = noFennecs && i === FENNECS_IDX;
    el.classList.toggle('na', isNA);
    if(!isNA) el.classList.toggle('off', !visible[i]);
    else el.classList.remove('off');
  });
}

function setTabs(cIdx){
  const d=DATA[cIdx];
  const btns=document.querySelectorAll('.tv');
  btns.forEach((b,i)=>{
    const disabled=
      (d.onlyT1&&i!==0)||
      (d.onlyT3&&i!==1)||
      (d.t[i]==null);
    b.disabled=disabled;
    if(disabled&&tIdx===i) tIdx=d.onlyT3?1:0;
  });
  btns.forEach(b=>b.classList.toggle('on',parseInt(b.dataset.t)===tIdx&&!b.disabled));
}

document.getElementById('cats').addEventListener('click',e=>{
  const b=e.target.closest('.cat');if(!b)return;
  document.querySelectorAll('.cat').forEach(x=>x.classList.remove('on'));
  b.classList.add('on');
  catIdx=parseInt(b.dataset.c);
  if(DATA[catIdx].onlyT3)tIdx=1;
  else if(DATA[catIdx].onlyT1)tIdx=0;
  setTabs(catIdx);
  syncLegendUI();
  buildChart();
});

document.getElementById('tvars').addEventListener('click',e=>{
  const b=e.target.closest('.tv');if(!b||b.disabled)return;
  document.querySelectorAll('.tv').forEach(x=>x.classList.remove('on'));
  b.classList.add('on');
  tIdx=parseInt(b.dataset.t);
  buildChart();
});

document.getElementById('leg').addEventListener('click',e=>{
  const el=e.target.closest('.li');
  if(!el||el.classList.contains('na'))return;
  const i=parseInt(el.dataset.i);
  visible[i]=!visible[i];
  syncLegendUI();
  if(chart){
    chart.data.datasets[i].hidden=!visible[i];
    chart.update();
  }
});

setTabs(0);
syncLegendUI();
buildChart();
